import type { ChatRequestData, ChatResponse } from '../../../contracts/modelGateway.js'

export type ConsistencyIssueCode = 'missing_goal' | 'missing_opinion' | 'unsupported_fact' | 'state_claim' | 'duplicate_message'
export interface ConsistencyIssue { code: ConsistencyIssueCode; reason: string }
export type ConsistencyError = 'budget_exhausted' | 'timeout' | 'invalid_response' | 'unavailable' | 'invalid_input'
export interface ConsistencyVerdict { verdict: 'accept' | 'revise'; issues: ConsistencyIssue[]; error?: ConsistencyError }
export interface ConsistencyInput {
  toolName: string
  candidate: Record<string, unknown>
  state: unknown
  sources: unknown
  sentMessages?: string[]
}
export interface ConsistencyReviewEvent {
  attempt: number
  status: 'accepted' | 'revised' | 'error' | 'budget_exhausted'
  ms: number
  usage: Record<string, unknown> | null
  model: string | null
  error?: ConsistencyError
}
export interface ConsistencyStats { checks: number; accepted: number; revised: number; errors: number; budgetExceeded: number }

function object(value: unknown): value is Record<string, unknown> {
  return !!value && typeof value === 'object' && !Array.isArray(value)
}

function parseVerdict(response: ChatResponse): ConsistencyVerdict {
  const message = response.choices?.[0]?.message
  if (response.error || !message || (Array.isArray(message.tool_calls) && message.tool_calls.length) || typeof message.content !== 'string' || message.content.length > 8192) throw new Error('invalid_response')
  const parsed: unknown = JSON.parse(message.content)
  if (!object(parsed) || Object.keys(parsed).some(key => !['verdict', 'issues'].includes(key)) || typeof parsed.verdict !== 'string' || !['accept', 'revise'].includes(parsed.verdict) || !Array.isArray(parsed.issues) || parsed.issues.length > 5) throw new Error('invalid_response')
  if ((parsed.verdict === 'accept') !== (parsed.issues.length === 0)) throw new Error('invalid_response')
  const issues: ConsistencyIssue[] = []
  for (const issue of parsed.issues) {
    if (!object(issue) || Object.keys(issue).some(key => !['code', 'reason'].includes(key)) || !ISSUE_CODES.has(issue.code as ConsistencyIssueCode) || typeof issue.reason !== 'string' || !issue.reason.trim() || issue.reason.length > 300) throw new Error('invalid_response')
    issues.push({ code: issue.code as ConsistencyIssueCode, reason: issue.reason.trim() })
  }
  return { verdict: parsed.verdict as ConsistencyVerdict['verdict'], issues }
}

/** One reviewer is scoped to one turn. It never writes state, sends, or retries. */
export function createCharacterConsistencyReviewer({ infer, maxChecks = 3, timeoutMs = 15_000, onResult }: {
  infer(request: ChatRequestData, signal: AbortSignal): Promise<ChatResponse>
  maxChecks?: number
  timeoutMs?: number
  onResult?(event: ConsistencyReviewEvent): void
}) {
  if (!Number.isSafeInteger(maxChecks) || maxChecks < 1 || maxChecks > 100) throw new Error('Invalid consistency check limit')
  if (!Number.isSafeInteger(timeoutMs) || timeoutMs < 1 || timeoutMs > 120_000) throw new Error('Invalid consistency timeout')
  const counters: ConsistencyStats = { checks: 0, accepted: 0, revised: 0, errors: 0, budgetExceeded: 0 }
  const emit = (event: ConsistencyReviewEvent) => { try { onResult?.(event) } catch { /* Telemetry cannot change the review outcome. */ } }
  async function review(input: ConsistencyInput): Promise<ConsistencyVerdict> {
    if (counters.checks >= maxChecks) {
      counters.budgetExceeded++
      emit({ attempt: counters.checks, status: 'budget_exhausted', ms: 0, usage: null, model: null, error: 'budget_exhausted' })
      return { verdict: 'revise', issues: [], error: 'budget_exhausted' }
    }
    const attempt = ++counters.checks, started = Date.now()
    let payload: string
    try {
      payload = JSON.stringify(input)
      if (!payload || Buffer.byteLength(payload, 'utf8') > 98_304) throw new Error('invalid_input')
    } catch {
      counters.errors++
      emit({ attempt, status: 'error', ms: Date.now() - started, usage: null, model: null, error: 'invalid_input' })
      return { verdict: 'revise', issues: [], error: 'invalid_input' }
    }
    const request: ChatRequestData = { messages: [{ role: 'system', content: SYSTEM }, { role: 'user', content: payload }], temperature: 0, max_tokens: 900 }
    const controller = new AbortController()
    let timer: ReturnType<typeof setTimeout> | undefined, timedOut = false, response: ChatResponse | undefined
    let failure: ConsistencyError = 'unavailable'
    try {
      response = await Promise.race([
        Promise.resolve().then(() => infer(request, controller.signal)),
        new Promise<never>((_resolve, reject) => {
          timer = setTimeout(() => { timedOut = true; controller.abort(); reject(new Error('timeout')) }, timeoutMs)
        }),
      ])
      failure = 'invalid_response'
      const verdict = parseVerdict(response)
      counters[verdict.verdict === 'accept' ? 'accepted' : 'revised']++
      emit({ attempt, status: verdict.verdict === 'accept' ? 'accepted' : 'revised', ms: Date.now() - started,
        usage: object(response.usage) ? response.usage : null, model: typeof response.model === 'string' ? response.model.slice(0, 200) : null })
      return verdict
    } catch {
      const error = timedOut ? 'timeout' : failure
      counters.errors++
      emit({ attempt, status: 'error', ms: Date.now() - started,
        usage: object(response?.usage) ? response.usage : null, model: typeof response?.model === 'string' ? response.model.slice(0, 200) : null, error })
      return { verdict: 'revise', issues: [], error }
    } finally {
      if (timer !== undefined) clearTimeout(timer)
    }
  }
  return { review, stats: (): ConsistencyStats => ({ ...counters }) }
}
